│       ├── GuideCodeBlock.jsx             # 소스 코드 블록

├── doc/                                   # 가이드에 렌더링되는 문서 콘텐츠

│   │   └── design-tokens.jsx
│   └── data/                              # 문서별 Props·클래스·토큰·예시 데이터
